%%% TABLE WIDTHS. \columnwidth is 240 pt in sigconf (8.5in paper, inner=outer=54pt,
%%%   columnsep=2pc). Seven single-column tables overran it. They are now set in
%%%   tabularx over \columnwidth with a ragged-right L column that absorbs the slack:
%%%   the text column wraps instead of spilling into the neighbouring column, whatever
%%%   the font metrics turn out to be.
\newcolumntype{L}{>{\raggedright\arraybackslash}X}

%%% FLOAT PLACEMENT. Chapter 6 declares ten floats for less than one page of running
%%%   text, six of them full width. With [t] as the only allowed position and the stock
%%%   float counters, the queue could never drain and figures surfaced two or three
%%%   sections later. Two changes answer that: the floats now carry [tbp] (or [tp] when
%%%   starred, since standard LaTeX has no bottom area for those), and the counters below
%%%   let a page hold more of them.
%%%   A third change, \afterpage{\clearpage} at the end of each chapter carrying full-width
%%%   floats, was removed on 2026-09-21: it is incompatible with double-column floats and
%%%   aborted the build on "Float(s) lost". The float mass is unchanged: see the note at the
%%%   end of 06_Empirical_Evaluation.tex.
\setcounter{topnumber}{3}
\setcounter{bottomnumber}{2}
\setcounter{totalnumber}{4}
\setcounter{dbltopnumber}{2}
\renewcommand{\topfraction}{0.85}
\renewcommand{\dbltopfraction}{0.85}
\renewcommand{\bottomfraction}{0.5}
\renewcommand{\textfraction}{0.1}
\renewcommand{\floatpagefraction}{0.75}
\renewcommand{\dblfloatpagefraction}{0.75}
